    function pickDistinctSongsForBook({ id, title, author, primaryTopic, synopsis }) {
      const recent = new Set(getRecentSongs().map((s) => s.toLowerCase()));
      const seed = Math.abs(
        `${id}-${title}-${author}-${primaryTopic}-${synopsis.slice(0, 40)}`
          .split("")
          .reduce((acc, ch) => acc + ch.charCodeAt(0), 0)
      );

      const pools = [
        { key: "innerConflict", list: SONG_POOLS_BY_ANGLE.innerConflict },
        { key: "settingAndMood", list: SONG_POOLS_BY_ANGLE.settingAndMood },
        { key: "characterAndChoice", list: SONG_POOLS_BY_ANGLE.characterAndChoice },
        { key: "turningPoint", list: SONG_POOLS_BY_ANGLE.turningPoint }
      ];

      const chosen = [];
      const usedTitles = new Set();

      pools.forEach((pool, pIdx) => {
        const ordered = [];
        for (let i = 0; i < pool.list.length; i++) {
          ordered.push(pool.list[(seed + pIdx * 3 + i) % pool.list.length]);
        }
        const nonRecent = ordered.find(
          (item) => !recent.has(`${item.title.toLowerCase()} — ${item.artist.toLowerCase()}`) && !usedTitles.has(item.title.toLowerCase())
        );
        const fallback = ordered.find((item) => !usedTitles.has(item.title.toLowerCase())) || ordered[0];
        const pick = nonRecent || fallback;
        usedTitles.add(pick.title.toLowerCase());
        chosen.push({
          title: `${pick.title} — ${pick.artist}`,
          storyAngle: pick.angle,
          note: `${pick.why} (Connected to "${title}"'s ${primaryTopic.toLowerCase()} arc.)`,
          url: `https://open.spotify.com/search/${encodeURIComponent(`${pick.title} ${pick.artist}`)}`
        });
      });

      recordRecentSongs(chosen.map((c) => {
        const [t, a] = c.title.split(" — ");
        return { title: t, artist: a || "" };
      }));

      return chosen;
    }

    function normalizePin(raw) {
      return String(raw || "").replace(/\D/g, "").slice(0, 4);

    function cleanHtml(str) {
      return String(str || "")
        .replace(/<[^>]*>/g, " ")
        .replace(/&amp;/g, "&")
        .replace(/&quot;/g, '"')
        .replace(/&#39;/g, "'")
        .replace(/\s+/g, " ")
        .trim();
    }

    function extractIsbn(raw) {
      const digits = String(raw || "").replace(/[^0-9Xx]/g, "").toUpperCase();
      return digits.length === 10 || digits.length === 13 ? digits : "";
    }

    async function queryOpenLibrary(query, scope) {
      const trimmed = query.trim();
      const isbn = extractIsbn(trimmed);
      if (isbn || scope === "isbn") {
        const code = isbn || trimmed.replace(/[^0-9Xx]/g, "");
        if (code) {
          const res = await fetch(`https://openlibrary.org/isbn/${encodeURIComponent(code)}.json`);
          if (res.ok) {
            const ed = await res.json();
            const desc = typeof ed.description === "string" ? ed.description : ed.description?.value || "";
            const coverId = Array.isArray(ed.covers) && ed.covers.length ? ed.covers[0] : null;
            return {
              title: ed.title || `ISBN ${code}`,
              subtitle: ed.subtitle || "",
              author: "Author on Edition",
              year: ed.publish_date || "",
              pages: ed.number_of_pages || null,
              isbn: code,
              subjects: Array.isArray(ed.subjects) ? ed.subjects.slice(0, 5) : [],
              description: cleanHtml(desc),
              coverUrl: coverId ? `https://covers.openlibrary.org/b/id/${coverId}-L.jpg` : `https://covers.openlibrary.org/b/isbn/${encodeURIComponent(code)}-L.jpg`,
              source: "Open Library",
              catalogUrl: `https://openlibrary.org/isbn/${encodeURIComponent(code)}`
            };
          }
        }
      }
      const params = new URLSearchParams({ limit: "6" });
      if (scope === "title") params.set("title", trimmed);
      else if (scope === "author") params.set("author", trimmed);
      else params.set("q", trimmed);
      const res = await fetch(`https://openlibrary.org/search.json?${params.toString()}`);
      if (!res.ok) return null;
      const data = await res.json();
      const doc = Array.isArray(data.docs) ? data.docs[0] : null;
      if (!doc || !doc.title) return null;
      const isbnGuess = Array.isArray(doc.isbn) && doc.isbn.length ? doc.isbn[0] : "";
      return {
        title: doc.title,
        subtitle: doc.subtitle || "",
        author: Array.isArray(doc.author_name) && doc.author_name.length ? doc.author_name.slice(0, 2).join(", ") : "Unknown Author",
        year: doc.first_publish_year ? String(doc.first_publish_year) : "",
        pages: doc.number_of_pages_median || null,
        isbn: isbnGuess,
        subjects: Array.isArray(doc.subject) ? doc.subject.slice(0, 5) : [],
        description: "",
        coverUrl: doc.cover_i ? `https://covers.openlibrary.org/b/id/${doc.cover_i}-L.jpg` : (isbnGuess ? `https://covers.openlibrary.org/b/isbn/${encodeURIComponent(isbnGuess)}-L.jpg` : ""),
        source: "Open Library",
        catalogUrl: doc.key ? `https://openlibrary.org${doc.key}` : "https://openlibrary.org"
      };
    }

    async function queryGoogleBooks(query, scope) {
      const trimmed = query.trim();
      const isbn = extractIsbn(trimmed);
      let q = trimmed;
      if (isbn || scope === "isbn") q = `isbn:${isbn || trimmed.replace(/[^0-9Xx]/g, "")}`;
      else if (scope === "title") q = `intitle:${trimmed}`;
      else if (scope === "author") q = `inauthor:${trimmed}`;
      const res = await fetch(`https://www.googleapis.com/books/v1/volumes?q=${encodeURIComponent(q)}&maxResults=1`);
      if (!res.ok) return null;
      const data = await res.json();
      const info = data?.items?.[0]?.volumeInfo;
      if (!info || !info.title) return null;
      const ids = Array.isArray(info.industryIdentifiers) ? info.industryIdentifiers : [];
      const isbn13 = ids.find((x) => x.type === "ISBN_13")?.identifier;
      const isbn10 = ids.find((x) => x.type === "ISBN_10")?.identifier;
      const thumb = info.imageLinks?.thumbnail || info.imageLinks?.smallThumbnail || "";
      return {
        title: info.title,
        subtitle: info.subtitle || "",
        author: Array.isArray(info.authors) && info.authors.length ? info.authors.slice(0, 2).join(", ") : "Unknown Author",
        year: info.publishedDate ? String(info.publishedDate).slice(0, 4) : "",
        pages: info.pageCount || null,
        isbn: isbn13 || isbn10 || isbn || "",
        subjects: Array.isArray(info.categories) ? info.categories.slice(0, 5) : [],
        description: cleanHtml(info.description || ""),
        coverUrl: thumb.replace(/^http:\/\//i, "https://"),
        source: "Google Books",
        catalogUrl: info.infoLink || "https://books.google.com"
      };
    }

    function combineRecords(olRec, gbRec) {
      if (!olRec && !gbRec) return null;
      const a = gbRec || {};
      const b = olRec || {};
      const description = (a.description && a.description.length > 50 ? a.description : "") || (b.description && b.description.length > 50 ? b.description : "") || a.description || b.description || "";
      const subjects = Array.from(new Set([...(a.subjects || []), ...(b.subjects || [])].filter(Boolean))).slice(0, 5);
      return {
        title: a.title || b.title || "Untitled Book",
        subtitle: a.subtitle || b.subtitle || "",
        author: (a.author && a.author !== "Unknown Author" ? a.author : "") || b.author || "Unknown Author",
        year: a.year || b.year || "",
        pages: a.pages || b.pages || null,
        isbn: a.isbn || b.isbn || "",
        subjects,
        description,
        coverUrl: a.coverUrl || b.coverUrl || "",
        source: olRec && gbRec ? "Open Library + Google Books" : (gbRec || olRec).source,
        catalogUrl: (gbRec || olRec).catalogUrl || "https://openlibrary.org"
      };
    }

    async function fetchWikipediaBookSubstance(title, author = "") {
      const cleanT = String(title || "").trim();
      if (!cleanT) return null;
      const cleanA = author && author !== "Unknown Author" ? String(author).trim() : "";
      const candidates = cleanA
        ? [`${cleanT} (${cleanA} novel)`, `${cleanT} (novel)`, `${cleanT} (${cleanA})`, cleanT]
        : [`${cleanT} (novel)`, `${cleanT} (book)`, cleanT];

      for (const pageTitle of candidates) {
        try {
          const res = await fetch(`https://en.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(pageTitle)}`);
          if (!res.ok) continue;
          const data = await res.json();
          if (data?.type === "standard" && data.extract && data.extract.length > 80) {
            const lower = data.extract.toLowerCase();
            if (lower.includes("may refer to:") || lower.includes("most commonly refers to")) continue;
            return {
              extract: cleanHtml(data.extract),
              wikiUrl: data.content_urls?.desktop?.page || `https://en.wikipedia.org/wiki/${encodeURIComponent(pageTitle.replace(/\s+/g, "_"))}`,
              thumbUrl: data.thumbnail?.source ? data.thumbnail.source.replace(/^http:\/\//i, "https://") : ""
            };
          }
        } catch (_) {}
      }
      return null;
    }

    async function enrichAnyBookRecord(record) {
      let enriched = { ...record };
      const wiki = await fetchWikipediaBookSubstance(enriched.title, enriched.author);
      if (wiki) {
        enriched.wikiSummary = wiki.extract;
        enriched.wikiUrl = wiki.wikiUrl;
        if (!enriched.description || enriched.description.length < wiki.extract.length) {
          enriched.description = wiki.extract;
        }
        if (!enriched.coverUrl && wiki.thumbUrl) enriched.coverUrl = wiki.thumbUrl;
      }
      if (!enriched.description || enriched.description.length < 70 || !enriched.pages) {
        try {
          const [olExtra, gbExtra] = await Promise.allSettled([
            queryOpenLibrary(enriched.isbn || enriched.title, enriched.isbn ? "isbn" : "title"),
            queryGoogleBooks(enriched.isbn ? `isbn:${enriched.isbn}` : `${enriched.title} ${enriched.author || ""}`.trim(), "all")
          ]);
          const combined = combineRecords(
            olExtra.status === "fulfilled" ? olExtra.value : null,
            gbExtra.status === "fulfilled" ? gbExtra.value : null
          );
          if (combined) {
            enriched = {
              ...enriched,
              description: (enriched.description && enriched.description.length >= 70 ? enriched.description : "") || combined.description || enriched.description,
              pages: enriched.pages || combined.pages,
              year: enriched.year || combined.year,
              subjects: enriched.subjects?.length ? enriched.subjects : combined.subjects,
              coverUrl: enriched.coverUrl || combined.coverUrl,
              catalogUrl: combined.catalogUrl || enriched.catalogUrl
            };
          }
        } catch (_) {}
      }
      return enriched;
    }

    async function searchCandidateBooks(query, scope = "all") {
      const trimmed = String(query || "").trim();
      const isbn = extractIsbn(trimmed);
      const candidates = [];
      const seen = new Set();

      try {
        const params = new URLSearchParams({ limit: "6" });
        if (isbn || scope === "isbn") params.set("q", isbn || trimmed);
        else if (scope === "title") params.set("title", trimmed);
        else if (scope === "author") params.set("author", trimmed);
        else params.set("q", trimmed);

        const res = await fetch(`https://openlibrary.org/search.json?${params.toString()}`);
        if (res.ok) {
          const data = await res.json();
          (data.docs || []).slice(0, 6).forEach((doc) => {
            if (!doc?.title) return;
            const authorStr = Array.isArray(doc.author_name) && doc.author_name.length ? doc.author_name.slice(0, 2).join(", ") : "Unknown Author";
            const k = `${doc.title.toLowerCase()}::${authorStr.toLowerCase()}`;
            if (seen.has(k)) return;
            seen.add(k);
            const isbnGuess = Array.isArray(doc.isbn) && doc.isbn.length ? doc.isbn[0] : "";
            candidates.push({
              title: doc.title,
              subtitle: doc.subtitle || "",
              author: authorStr,
              year: doc.first_publish_year ? String(doc.first_publish_year) : "",
              pages: doc.number_of_pages_median || null,
              isbn: isbnGuess,
              subjects: Array.isArray(doc.subject) ? doc.subject.slice(0, 5) : [],
              description: "",
              coverUrl: doc.cover_i ? `https://covers.openlibrary.org/b/id/${doc.cover_i}-M.jpg` : (isbnGuess ? `https://covers.openlibrary.org/b/isbn/${encodeURIComponent(isbnGuess)}-M.jpg` : ""),
              source: "Open Library",
              catalogUrl: doc.key ? `https://openlibrary.org${doc.key}` : "https://openlibrary.org"
            });
          });
        }
      } catch (_) {}

      return candidates.slice(0, 8);
    }

    function buildStudyKit({ title, author, year, primaryTopic, subjects, leadSentence }) {
      const cleanAuthor = author && author !== "Unknown Author" ? author : "Author";
      const parts = cleanAuthor.trim().split(/\s+/);
      const lastName = parts.length > 1 ? parts[parts.length - 1] : cleanAuthor;
      const firstInitial = parts.length > 1 ? `${parts[0][0]}.` : "";
      const mlaAuthor = parts.length > 1 ? `${lastName}, ${parts.slice(0, -1).join(" ")}` : cleanAuthor;
      const apaAuthor = firstInitial ? `${lastName}, ${firstInitial}` : cleanAuthor;
      const cleanYear = /^\d{4}$/.test(String(year).trim()) ? String(year).trim() : "n.d.";

      const mla = `${mlaAuthor}. *${title}*. ${cleanYear !== "n.d." ? `${cleanYear}.` : ""}`.trim();
      const apa = `${apaAuthor} (${cleanYear}). *${title}*.`;
      const topicFocus = subjects.slice(0, 2).join(" and ").toLowerCase() || primaryTopic.toLowerCase();

      const essayAngles = [
        `How does "${title}" handle ${topicFocus}, and where does the author's perspective feel most deliberate?`,
        `Compare how the opening setup (“${leadSentence.slice(0, 85)}${leadSentence.length > 85 ? "…" : ""}”) connects to the main tension in ${primaryTopic}.`,
        `Pick one choice made by a central figure or argument in "${title}" and argue whether it solves or deepens the core conflict.`
      ];

      return {
        mla,
        apa,
        essayAngles,
        recallQuestion: `In 2–3 sentences from memory (without looking at the blurb), what is the main focus or conflict in "${title}" so far?`,
        recallHint: `Check your answer against: ${primaryTopic} • ${leadSentence}`
      };
    }

    function buildDynamicCompanion(record) {
      const title = cleanHtml(record.title) || "Untitled Book";
      const subtitle = cleanHtml(record.subtitle);
      const author = cleanHtml(record.author) || "Unknown Author";
      const year = cleanHtml(record.year) || "Edition Listed";
      const pages = record.pages && Number(record.pages) > 0 ? Number(record.pages) : 250;
      const subjects = Array.isArray(record.subjects) && record.subjects.length
        ? record.subjects.map(cleanHtml).filter(Boolean)
        : ["General"];
      const primaryTopic = subjects[0] || "General";
      const rawDesc = cleanHtml(record.description);
      const synopsis = rawDesc || `"${title}"${author !== "Unknown Author" ? ` by ${author}` : ""} (${primaryTopic}). No full blurb was in the catalog record, so jot down what happens as you read.`;

      const sentences = synopsis.split(/(?<=[.!?])\s+/).map((s) => s.trim()).filter(Boolean);
      const leadSentence = sentences[0] || `"${title}" sets up its main situation early on.`;
      const followSentence = sentences.slice(1, 3).join(" ") || "Watch how the main characters react when plans change.";
      const topicText = subjects.slice(0, 2).join(" & ") || primaryTopic;
      const authorLabel = author !== "Unknown Author" ? author : "its author";
      const yearLabel = year && year !== "Edition Listed" ? ` (${year})` : "";

      const whyRead = `"${title}" is worth picking up if you want to explore ${topicText.toLowerCase()} through a clear central situation${author !== "Unknown Author" ? ` shaped by ${author}` : ""}. It gives students strong material for reflection, discussion, and comparing how big ideas play out in concrete choices.`;

      const characters = [
        {
          name: `Central Perspective / Lead in "${title}"`,
          role: "Focal Point & Agency",
          breakdown: `${leadSentence} Trace what this figure risks or values most when pressure builds.`
        },
        {
          name: `Key Tension & Counter-Force (${primaryTopic})`,
          role: "Conflict & Turning Point",
          breakdown: `${followSentence || `Observe how external expectations or obstacles in "${title}" test the protagonist's assumptions.`}`
        }
      ];

      const deepAnalysis = [
        `Central Tension: "${title}" brings ${topicText.toLowerCase()} into direct friction with the characters' immediate choices and surroundings.`,
        `Themes & Motifs: Pay attention to how ${authorLabel} repeats key images, choices, or arguments around "${leadSentence.slice(0, 100)}${leadSentence.length > 100 ? "…" : ""}".`,
        `Character Psychology & Teen Takeaway: Notice what shifts internally when plans falter—this is where the sharpest essay claims and personal takeaways live for young readers.`
      ].join(" ");

      const takeaways = [
        `Core focus: ${topicText} in "${title}" by ${authorLabel}${yearLabel}.`,
        sentences.length > 1 ? `Key plot/idea thread: ${sentences[1]}` : `Read how "${title}" develops its central situation across ~${pages} pages.`,
        `Reading takeaway: Track how the central tension shifts from the opening setup to the middle turning point.`
      ];

      const quotes = [
        { tag: "Opening Setup / Premise", quote: leadSentence, note: `Ask what question or expectation "${title}" sets up right here.` },
        { tag: "Developing Thread", quote: followSentence || leadSentence, note: `Watch how ${authorLabel} builds stakes or evidence around ${primaryTopic.toLowerCase()}.` }
      ];

      const storyNotes = {
        arc: `${leadSentence} ${followSentence}`.trim().slice(0, 440),
        themes: `Centered on ${subjects.slice(0, 3).join(", ").toLowerCase() || primaryTopic.toLowerCase()}—pay attention to recurring motifs, choices, or arguments that evolve across the book.`,
        pacing: `Roughly ${pages} pages${year && year !== "Edition Listed" ? ` (${year})` : ""}. Aim for ~15–25 pages per sitting and note where scenes or chapters accelerate vs. pause for reflection.`
      };

      const studyKit = buildStudyKit({ title, author, year, primaryTopic, subjects, leadSentence });
      const id = `${title}-${author}-${record.isbn || ""}`.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "").slice(0, 60) || `book-${Date.now()}`;
      const spineColor = SPINE_COLORS[Math.abs(id.split("").reduce((acc, ch) => acc + ch.charCodeAt(0), 0)) % SPINE_COLORS.length];

      const tracks = pickDistinctSongsForBook({ id, title, author, primaryTopic, synopsis });

      return {
        id,
        title,
        subtitle,
        author,
        year,
        pages,
        isbn: record.isbn || "—",
        primaryTopic,
        synopsis,
        whyRead,
        characters,
        deepAnalysis,
        takeaways,
        quotes,
        storyNotes,
        studyKit,
        adaptationText: `Search for film, TV, or stage adaptations of "${title}"${author !== "Unknown Author" ? ` by ${author}` : ""} to see what changed from the page.`,
        adaptationSearchUrl: `https://www.youtube.com/results?search_query=${encodeURIComponent(`${title} ${author !== "Unknown Author" ? author : ""} adaptation trailer`)}`,
        moodTitle: `${primaryTopic} Soundtrack`,
        spotifySearchUrl: `https://open.spotify.com/search/${encodeURIComponent(`${title} ${primaryTopic} reading`)}`,
        tracks,
        coverUrl: record.coverUrl || "",
        wikiUrl: record.wikiUrl || "",
        source: record.source || "Quillfinite",
        catalogUrl: record.catalogUrl || `https://openlibrary.org/search?q=${encodeURIComponent(title)}`,
        spineColor
      };
    }

    async function enrichBookWithDeepAnalysisAndSongs(book) {
      const apiKey = getGeminiApiKey();
      const cacheKey = `${book.id}::${apiKey ? "gemini" : "dynamic"}`;
      const cached = getAiBookCache()[cacheKey];
      if (cached) {
        return { ...book, ...cached };
      }

      if (!apiKey) {
        const dynamicPatch = {
          tracks: pickDistinctSongsForBook(book)
        };
        saveAiBookCache(cacheKey, dynamicPatch);
        return { ...book, ...dynamicPatch };
      }

      const recentSongs = getRecentSongs().slice(0, 30);
      const systemInstruction = `
You are the literary analyst and soundtrack curator for Quillfinite, a teen reading app.
Generate a deep literary analysis and 4 distinct, book-specific songs for ANY book.
Rules:
1. Do not give a shallow plot summary alone; analyze central tension, themes/motifs, character psychology, and teen takeaway.
2. Choose 4 songs that genuinely connect to THIS book's themes, characters, conflicts, or setting.
3. Avoid repeating any song in this recently used list: [${recentSongs.join("; ") || "None"}].
4. Every song must include storyAngle and whyItRelates explaining how it connects to this specific book.
`.trim();

      const prompt = `
Book Title: ${book.title}
Author: ${book.author}
Genre / Topic: ${book.primaryTopic}
Synopsis / Notes: ${book.synopsis}
`.trim();

      try {
        const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${encodeURIComponent(apiKey)}`;
        const res = await fetch(endpoint, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            systemInstruction: { parts: [{ text: systemInstruction }] },
            contents: [{ role: "user", parts: [{ text: prompt }] }],
            generationConfig: {
              temperature: 0.85,
              responseMimeType: "application/json",
              responseSchema: BOOK_INSIGHT_SCHEMA
            }
          })
        });
        if (!res.ok) throw new Error(`Gemini error ${res.status}`);
        const data = await res.json();
        const rawText = data?.candidates?.[0]?.content?.parts?.[0]?.text;
        if (!rawText) throw new Error("Empty model response");
        const parsed = JSON.parse(rawText);

        const formattedTracks = (parsed.songs || []).slice(0, 4).map((s) => ({
          title: `${s.title} — ${s.artist}`,
          storyAngle: s.storyAngle || "Story Connection",
          note: s.whyItRelates || "",
          url: `https://open.spotify.com/search/${encodeURIComponent(`${s.title} ${s.artist}`)}`
        }));

        recordRecentSongs(parsed.songs || []);

        const combinedAnalysis = [
          parsed.centralTension ? `Central Tension: ${parsed.centralTension}` : "",
          parsed.themesAndMotifs ? `Themes & Motifs: ${parsed.themesAndMotifs}` : "",
          parsed.characterPsychology ? `Character Psychology: ${parsed.characterPsychology}` : "",
          parsed.teenTakeaway ? `Teen Takeaway: ${parsed.teenTakeaway}` : ""
        ].filter(Boolean).join(" ");

        const patch = {
          whyRead: parsed.whyRead || book.whyRead,
          deepAnalysis: combinedAnalysis || book.deepAnalysis,
          characters: Array.isArray(parsed.characters) && parsed.characters.length ? parsed.characters : book.characters,
          tracks: formattedTracks.length ? formattedTracks : book.tracks,
          source: `${book.source} + Gemini Analysis`
        };

        saveAiBookCache(cacheKey, patch);
        return { ...book, ...patch };
      } catch (_) {
        return { ...book, tracks: pickDistinctSongsForBook(book) };
      }
    }

    async function lookupBook(rawQuery, scope = "all", previewImageUrl = "") {
      const q = String(rawQuery || "").trim();

        return;
      }
      appState.lastQuery = q;
      document.getElementById("loadingBookCard").classList.remove("hidden");
      document.getElementById("emptyBookCard").classList.add("hidden");
      document.getElementById("activeBookReadingCard").classList.add("hidden");

        img.src = previewImageUrl;
        img.classList.remove("hidden");

        const base = buildDynamicCompanion({ title: q, coverUrl: previewImageUrl });
        const full = await enrichBookWithDeepAnalysisAndSongs(base);
        displayBook(full);
      }
    }

      document.getElementById("receiptProgressText").textContent = `p.${prog.cur} / ${prog.tot} (${pct}%)`;
      document.getElementById("receiptCurPage").value = prog.cur;
      document.getElementById("receiptTotPages").value = prog.tot;
      document.getElementById("receiptPctLabel").textContent = `${pct}%`;
      document.getElementById("receiptBarFill").style.width = `${pct}%`;

    }

    function renderBookSources(book) {
      const list = document.getElementById("bookSourcesList");
      const badge = document.getElementById("bookSourcesCountBadge");
      if (!list) return;
      const sources = [
        { label: `${book.source || "Book Catalog"} — Record for "${book.title}"`, url: book.catalogUrl },
        ...(book.wikiUrl ? [{ label: `Wikipedia — Overview & Context for "${book.title}"`, url: book.wikiUrl }] : []),
        { label: "Philippine Statistics Authority (PSA) — 2024 FLEMMS", url: "https://psa.gov.ph/" },
        { label: "World Bank — Philippines Learning Poverty Data", url: "https://www.worldbank.org/en/country/philippines" }
      ];
      if (badge) badge.textContent = `${sources.length} sources`;
      list.innerHTML = "";
      sources.forEach((src) => {
        const li = document.createElement("li");
        li.className = "flex flex-wrap items-center justify-between gap-2";
        li.innerHTML = `<span>• ${src.label}</span><a href="${src.url}" target="_blank" rel="noopener noreferrer" class="font-bold underline text-[#8b4b32] shrink-0">Source ↗</a>`;
        list.appendChild(li);
      });
    }

    function handleCorkboardPhoto(e) {
      const file = e.target.files?.[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        appState.pendingCorkPhoto = { dataUrl: reader.result, name: file.name };
        document.getElementById("corkPhotoPreviewImg").src = reader.result;
        document.getElementById("corkPhotoFileName").textContent = file.name;
        document.getElementById("corkPhotoPreviewWrap").classList.remove("hidden");
      };
      reader.readAsDataURL(file);
    }

    function clearCorkboardPhoto() {
      appState.pendingCorkPhoto = null;
      document.getElementById("corkPhotoPreviewWrap").classList.add("hidden");
    }

    function addCorkboardPin(e) {
      e.preventDefault();
      if (!appState.currentBook) return;
      const text = document.getElementById("corkNoteTextInput").value.trim();
      const page = document.getElementById("corkPageTagInput").value.trim();
      const photo = appState.pendingCorkPhoto ? appState.pendingCorkPhoto.dataUrl : "";
      if (!text && !photo) {
        showBanner("Write a quote/note or attach a picture to pin.");
        return;
      }
      const id = appState.currentBook.id;
      const size = document.getElementById("corkPinSizeSelect")?.value || "standard";
      appState.corkPins[id] = [{ text: text || "Pinned photo note", page, photo, size }, ...(appState.corkPins[id] || [])];
      document.getElementById("corkNoteTextInput").value = "";
      document.getElementById("corkPageTagInput").value = "";
      clearCorkboardPhoto();
      renderBookCorkboard(id);
      persistActiveAccountData();
      showBanner("Pinned item to the corkboard!");
    }

    function renderBookCorkboard(bookId) {
      const pins = appState.corkPins[bookId] || [];
      document.getElementById("corkPinCountBadge").textContent = `${pins.length} ${pins.length === 1 ? "pin" : "pins"}`;
      const grid = document.getElementById("bookCorkboardGrid");
      grid.innerHTML = "";
      if (!pins.length) {
        grid.innerHTML = `<p class="sm:col-span-2 text-xs bg-[#fffdf5]/90 rounded-xl p-3">No pins on this book's corkboard yet—add a quote or picture above.</p>`;
        return;
      }
      const pinColors = ["bg-rose-500", "bg-amber-400", "bg-emerald-600", "bg-sky-500"];
      pins.forEach((p, idx) => {
        const curSize = p.size || "standard";
        const sizeClass = curSize === "feature" ? "cork-size-feature" : curSize === "compact" ? "cork-size-compact" : "cork-size-standard";
        const card = document.createElement("div");
        card.className = `cork-pin-card ${sizeClass} p-3 pt-6 relative`;
        card.innerHTML = `
          <span class="pushpin-head ${pinColors[idx % pinColors.length]} absolute top-2 left-1/2 -translate-x-1/2"></span>
          <div class="absolute top-1.5 right-2 flex items-center gap-1">
            <select aria-label="Pin size" class="text-[10px] rounded bg-amber-50 px-1 py-0.5 border border-stone-800/30">
              <option value="compact" ${curSize === "compact" ? "selected" : ""}>S</option>
              <option value="standard" ${curSize === "standard" ? "selected" : ""}>M</option>
              <option value="feature" ${curSize === "feature" ? "selected" : ""}>L</option>
            </select>
            <button type="button" class="text-xs font-bold text-stone-500 hover:text-rose-700" title="Remove">✕</button>
          </div>
          ${p.photo ? `<div class="polaroid-frame mb-2"><img src="${p.photo}" alt="Pinned photo" /></div>` : ""}
          ${p.page ? `<span class="inline-block text-[11px] font-bold px-2 py-0.5 rounded-full bg-amber-100 mb-1">${p.page}</span>` : ""}
          <p class="${curSize === "feature" ? "text-base font-medium" : curSize === "compact" ? "text-xs" : "text-sm"} text-stone-800">${p.text}</p>
        `;
        card.querySelector("select").onchange = (ev) => {
          p.size = ev.target.value;
          renderBookCorkboard(bookId);
          persistActiveAccountData();
        };
        card.querySelector("button").onclick = () => {
          appState.corkPins[bookId] = pins.filter((_, i) => i !== idx);
          renderBookCorkboard(bookId);
          persistActiveAccountData();
        };
        grid.appendChild(card);
      });
    }

    function toggleCurrentBookShelf() {
      const book = appState.currentBook;
      if (!book) return;
      const exists = appState.savedBooks.some((b) => b.id === book.id);
      appState.savedBooks = exists ? appState.savedBooks.filter((b) => b.id !== book.id) : [book, ...appState.savedBooks];
      document.getElementById("headerShelfCount").textContent = String(appState.savedBooks.length);
      persistActiveAccountData();
      displayBook(book);
      renderBookshelf();
      showBanner(exists ? `Removed "${book.title}" from My Bookshelf.` : `Added "${book.title}" to My Bookshelf!`);
    }

    function toggleCurrentBookFinished() {
      const book = appState.currentBook;
      if (!book) return;
      const done = appState.finishedIds.includes(book.id);
      appState.finishedIds = done ? appState.finishedIds.filter((x) => x !== book.id) : [...appState.finishedIds, book.id];
      persistActiveAccountData();
      displayBook(book);
    }

      renderBookshelf();
    }

    function renderBookshelf() {
      const row = document.getElementById("shelfSpinesRow");
      const list = document.getElementById("shelfCardsList");
      row.className = `wood-shelf ${appState.shelfTheme} rounded-2xl p-4 sm:p-6 flex flex-wrap items-end gap-3 min-h-[200px]`;
      row.innerHTML = "";
      list.innerHTML = "";
      document.getElementById("shelfSummaryCount").textContent = `${appState.savedBooks.length} ${appState.savedBooks.length === 1 ? "book" : "books"} saved`;
      document.getElementById("headerShelfCount").textContent = String(appState.savedBooks.length);

      if (!appState.savedBooks.length) {
        row.innerHTML = `<p class="text-amber-50 text-sm">No books on your shelf yet—look up or scan any book in the Book Booth and tap “+ Add to Bookshelf”.</p>`;
        return;
      }

      appState.savedBooks.forEach((b) => {
        const spine = document.createElement("button");
        spine.type = "button";
        spine.className = "book-spine text-sm text-amber-50";
        spine.style.backgroundColor = b.spineColor || "#c76e4b";
        spine.textContent = b.title;
        spine.onclick = () => { showSection("booth"); displayBook(b); };
        row.appendChild(spine);

        const card = document.createElement("div");
        card.className = "rounded-2xl bg-[#fffaf0] p-4 flex flex-col justify-between gap-2.5";
        card.innerHTML = `
          <div>
            <span class="text-xs px-2 py-0.5 sketch-pill bg-amber-100">${b.primaryTopic}</span>
            <h3 class="font-display text-lg font-bold mt-1.5">${b.title}</h3>
            <p class="text-xs text-stone-600">by ${b.author}</p>
            <p class="text-xs text-stone-700 mt-1.5 line-clamp-2">${b.synopsis}</p>
          </div>
          <button type="button" class="sketch-btn bg-white px-3 py-1 text-xs font-semibold self-start">Open Card &amp; Notes →</button>
        `;
        card.querySelector("button").onclick = () => { showSection("booth"); displayBook(b); };
        list.appendChild(card);
      });
    }

      const wrap = document.getElementById("cupButtonsWrap");
      wrap.innerHTML = "";
      CUPS.forEach((c) => {
        const active = c.name === appState.activeCup;
        const btn = document.createElement("button");

        btn.textContent = c.name;

    function renderStudyStudio() {
      const m = String(Math.floor(appState.timerSec / 60)).padStart(2, "0");
      const s = String(appState.timerSec % 60).padStart(2, "0");
      document.getElementById("studioTimerClock").textContent = `${m}:${s}`;
      const ul = document.getElementById("studioTaskList");
      ul.innerHTML = "";
      appState.tasks.forEach((t) => {
        const li = document.createElement("li");
        li.className = "flex items-center gap-2";
        li.innerHTML = `<input type="checkbox" ${t.done ? "checked" : ""} /><span class="${t.done ? "line-through text-stone-400" : ""}">${t.text}</span>`;
        li.querySelector("input").onchange = () => { t.done = !t.done; persistActiveAccountData(); renderStudyStudio(); };
        ul.appendChild(li);
      });
      document.getElementById("studioPromptText").textContent = `“${REFLECTION_PROMPTS[appState.promptIdx % REFLECTION_PROMPTS.length]}”`;
      const pagesLeft = Math.max(1, Number(document.getElementById("pacePagesLeftInput")?.value) || 180);
      const daysLeft = Math.max(1, Number(document.getElementById("paceDaysLeftInput")?.value) || 6);
      const perDay = Math.ceil(pagesLeft / daysLeft);
      document.getElementById("paceResultText").textContent = `${perDay} ${perDay === 1 ? "page" : "pages"} / day`;
      document.getElementById("paceSubtext").textContent = `~${Math.max(10, Math.round(perDay * 1.4))} mins/day across ${daysLeft} ${daysLeft === 1 ? "day" : "days"} (${pagesLeft} pages left).`;
    }

    function addPacePlanToChecklist() {
      const pagesLeft = Math.max(1, Number(document.getElementById("pacePagesLeftInput")?.value) || 180);
      const daysLeft = Math.max(1, Number(document.getElementById("paceDaysLeftInput")?.value) || 6);
      const perDay = Math.ceil(pagesLeft / daysLeft);
      appState.tasks.unshift({ id: Date.now(), text: `Read ${perDay} ${perDay === 1 ? "page" : "pages"} today`, done: false });
      persistActiveAccountData();
      renderStudyStudio();
      showBanner(`Added today's ${perDay}-page goal to your Study Checklist.`);
    }

    function addStudioTask(e) {
      e.preventDefault();
      const inp = document.getElementById("studioTaskInput");
      if (!inp.value.trim()) return;
      appState.tasks.push({ id: Date.now(), text: inp.value.trim(), done: false });
      inp.value = "";
      persistActiveAccountData();
      renderStudyStudio();
    }

    function nextStudioPrompt() {
      appState.promptIdx += 1;
      renderStudyStudio();
    }

    function toggleStudioTimer() {
      const btn = document.getElementById("studioTimerBtn");
      if (appState.timerRunning) {
        clearInterval(studioTimerId);
        studioTimerId = null;
        appState.timerRunning = false;
        btn.textContent = "Start";
      } else {
        appState.timerRunning = true;
        btn.textContent = "Pause";
        studioTimerId = setInterval(() => {
          if (appState.timerSec > 0) {
            appState.timerSec -= 1;
            renderStudyStudio();
          } else {
            clearInterval(studioTimerId);
            studioTimerId = null;
            appState.timerRunning = false;
            btn.textContent = "Start";
            showBanner("Focus timer finished!");
          }
        }, 1000);
      }
    }

    function resetStudioTimer() {
      clearInterval(studioTimerId);
      studioTimerId = null;
      appState.timerRunning = false;
      appState.timerSec = 25 * 60;
      document.getElementById("studioTimerBtn").textContent = "Start";
      renderStudyStudio();
    }
